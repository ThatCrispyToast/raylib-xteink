cmake_minimum_required(VERSION 3.20)
project(raylib_x4 VERSION 1.1.1 LANGUAGES C CXX)
option(X4_SANITIZE "Build with address and undefined behavior sanitizers" OFF)
option(X4_BUILD_EMULATOR "Build the microreader X4 emulator adapter" ON)
set(X4_MICROREADER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/research/microreader" CACHE PATH "Pinned microreader checkout")
set(X4_APP_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/examples/paper.c" CACHE FILEPATH "Application entry point")
set(X4_APP_EXTRA_SOURCES "" CACHE STRING "Additional application sources (semicolon-separated)")
include(cmake/raylib-x4.cmake)
if(X4_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(raylib_x4 STATIC ${X4_RAYLIB_SOURCES})
add_library(raylib::x4 ALIAS raylib_x4)
x4_configure(raylib_x4)
target_link_libraries(raylib_x4 PUBLIC m)
target_link_options(raylib_x4 INTERFACE -Wl,--gc-sections)
if(X4_BUILD_EMULATOR)
    find_package(SDL2 REQUIRED)
    if(NOT EXISTS "${X4_MICROREADER_DIR}/platforms/desktop/display.h")
        message(FATAL_ERROR "Run python3 scripts/fetch-emulator.py first")
    endif()
    add_library(x4_emulator STATIC src/x4_hal_emulator.cpp)
    target_include_directories(x4_emulator PRIVATE src include
        ${X4_MICROREADER_DIR}/platforms/desktop ${X4_MICROREADER_DIR}/lib)
    target_link_libraries(x4_emulator PUBLIC SDL2::SDL2)
    target_compile_features(x4_emulator PRIVATE cxx_std_17)
    add_executable(x4_demo ${X4_APP_SOURCE} ${X4_APP_EXTRA_SOURCES})
    if(X4_APP_SOURCE STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}/examples/paper.c")
        target_sources(x4_demo PRIVATE examples/paper_state.c)
    endif()
    target_link_libraries(x4_demo PRIVATE raylib_x4 x4_emulator)
    add_executable(x4_emulator_tests tests/test_emulator.c)
    target_link_libraries(x4_emulator_tests PRIVATE raylib_x4 x4_emulator)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME x4_package COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-package.py)
    add_library(raylib_x4_reference STATIC ${X4_RAYLIB_SOURCES})
    x4_configure(raylib_x4_reference)
    target_compile_definitions(raylib_x4_reference PRIVATE X4_DISABLE_FAST_RASTER)
    target_link_libraries(raylib_x4_reference PUBLIC m)
    add_executable(x4_raster_fast tests/test_raster_compare.c tests/hal_test.c)
    target_include_directories(x4_raster_fast PRIVATE src tests)
    target_link_libraries(x4_raster_fast PRIVATE raylib_x4)
    add_executable(x4_raster_reference tests/test_raster_compare.c tests/hal_test.c)
    target_include_directories(x4_raster_reference PRIVATE src tests)
    target_link_libraries(x4_raster_reference PRIVATE raylib_x4_reference)
    add_test(NAME x4_raster_equivalence COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-raster-compare.py ${CMAKE_CURRENT_BINARY_DIR})
    add_executable(paper_tests tests/test_paper.c examples/paper_state.c)
    target_include_directories(paper_tests PRIVATE examples)
    add_test(NAME paper_state COMMAND paper_tests)
    add_executable(x4_tests tests/test_x4.c tests/hal_test.c)
    target_include_directories(x4_tests PRIVATE src tests)
    target_link_libraries(x4_tests PRIVATE raylib_x4)
    add_test(NAME x4_core COMMAND x4_tests)
    add_executable(x4_epd_tests tests/test_epd.c src/x4_epd.c)
    target_include_directories(x4_epd_tests PRIVATE src include)
    add_test(NAME x4_epd COMMAND x4_epd_tests)
    if(X4_BUILD_EMULATOR AND X4_APP_SOURCE STREQUAL "${CMAKE_CURRENT_SOURCE_DIR}/examples/paper.c")
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
            add_test(NAME x4_input_timing COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-input-timing.py ${CMAKE_CURRENT_BINARY_DIR})
            set_tests_properties(x4_input_timing PROPERTIES TIMEOUT 60)
        endif()
        add_test(NAME x4_emulator COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-emulator.py ${CMAKE_CURRENT_BINARY_DIR})
    endif()
endif()
